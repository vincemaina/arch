# Treesitter grammars, from pacman rather than from a plugin that downloads and
# compiles them at runtime. Same argument that ruled out Mason for language
# servers: anything installed outside these manifests is invisible here and
# absent from a rebuilt machine.
#
# Nvim itself ships c, lua, markdown, query, vim and vimdoc. These are the ones
# the official repositories add that this machine's languages need.
#
# Missing and known: typescript, sql, html, css, json, yaml and toml have no
# package at all. See TASK-84, which is blocked on the AUR decision in TASK-43.
# Those filetypes fall back to vim's regex highlighting until then.
tree-sitter-python
tree-sitter-javascript
tree-sitter-bash

# Node, for the language servers that exist only as npm packages.
#
# html, css, json and emmet have no server in the official repositories. The AUR
# has them, and TASK-43 decided against the AUR - partly on its costs, partly
# because those particular packages are among the least maintained things in it:
# vscode-langservers-extracted has four votes and was last touched in May 2024,
# and its PKGBUILD is a wrapper around `npm install`.
#
# So npm directly, which removes a layer rather than adding one. What is
# installed is pinned by a lockfile tracked in this repository - see
# dot_local/lib/language-servers/ - so it is as reproducible as a package
# manifest and visible in exactly the same way.
nodejs
npm

# Language servers from the official repositories. The ones that are NOT here -
# html, css, json and emmet - have no package and come from npm instead; see
# dot_local/lib/language-servers/.
#
# pyright for types and navigation, ruff for linting and formatting. Two tools
# rather than one because they do different halves and neither does both well:
# pyright has no formatter and ruff has no type checker.
pyright
ruff

# javascript and typescript. The server is written in typescript and needs the
# typescript compiler package at runtime, which is a dependency rather than
# something to install separately.
typescript-language-server
typescript
